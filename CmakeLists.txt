cmake_minimum_required(VERSION 3.22)

project(TrailTorque LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)


# ---------------------------------------------------------
# THIRD-PARTY LIBRARIES
# ---------------------------------------------------------

add_subdirectory(
    ThirdParty/SDL
    EXCLUDE_FROM_ALL
)

add_subdirectory(
    ThirdParty/Box2D
    EXCLUDE_FROM_ALL
)

# Don't build SDL_ttf sample programs.
set(SDLTTF_SAMPLES OFF CACHE BOOL "" FORCE)

add_subdirectory(
    ThirdParty/SDL_ttf
    EXCLUDE_FROM_ALL
)


# ---------------------------------------------------------
# TRAILTORQUE
# ---------------------------------------------------------

add_executable(
    TrailTorque
    source/main.cpp
)


# ---------------------------------------------------------
# LINK LIBRARIES
# ---------------------------------------------------------

target_link_libraries(
    TrailTorque
    PRIVATE
        SDL3::SDL3
        SDL3_ttf::SDL3_ttf
        box2d::box2d
)


# ---------------------------------------------------------
# WINDOWS DLL COPYING
# ---------------------------------------------------------

if(WIN32)

    # SDL3.dll
    add_custom_command(
        TARGET TrailTorque
        POST_BUILD

        COMMAND
            "${CMAKE_COMMAND}" -E copy_if_different
            $<TARGET_FILE:SDL3::SDL3-shared>
            $<TARGET_FILE_DIR:TrailTorque>

        VERBATIM
    )


    # SDL3_ttf.dll
    if(TARGET SDL3_ttf-shared)

        add_custom_command(
            TARGET TrailTorque
            POST_BUILD

            COMMAND
                "${CMAKE_COMMAND}" -E copy_if_different
                $<TARGET_FILE:SDL3_ttf-shared>
                $<TARGET_FILE_DIR:TrailTorque>

            VERBATIM
        )

    endif()


    # Box2D.dll
    if(TARGET box2d::box2d)

        add_custom_command(
            TARGET TrailTorque
            POST_BUILD

            COMMAND
                "${CMAKE_COMMAND}" -E copy_if_different
                $<TARGET_FILE:box2d::box2d>
                $<TARGET_FILE_DIR:TrailTorque>

            VERBATIM
        )

    endif()

endif()
